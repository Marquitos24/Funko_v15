Me meto en la carpeta del proyecto en la terminal y ejecuto --> dotnet user-secrets init (vínculo oculto en el ordenador)

Guardo un secreto con un comando --> dotnet user-secrets set "Jwt:Secret" "mi-super-clave-secreta-que-nadie-debe-ver"

Se guarda en un archivo JSON del SO (EJ: Dentro de C:\Users\tu-usuario\AppData\Roaming\Microsoft\UserSecrets\...). 
Como no está dentro de tu proyecto, Git jamás lo subirá a GitHub.
 .NET lo lee Automáticamente. 
 Cuando la App arranque en modo Development, .NET mira en appsettings.json, 
 luego en appsettings.Development.json y también mira en tu almacén de User Secrets, 
 sobreescribiendo lo que haga falta de forma transparente.
 
 Al abrir la terminal (PowerShell) antes de ejecutar la app en producción y pones:
 $env:Jwt__Secret = "clave-de-produccion-ultrasegura"
 
 ASP.NET Core interceptará esa variable de entorno y se la asignará automáticamente a 
 Jwt:Secret en tu IConfiguration, sin que tengas que cambiar nada de código.
 
 
 
 INICIAR APP DESARROLLO ==> $env:ASPNETCORE_ENVIRONMENT = "Development"
                            dotnet run
  DOCKER-COMPOSE ==> docker compose -u up 
                            
 INICIAR APP PRODUCCION ==> $env:ASPNETCORE_ENVIRONMENT = "Production"
                            dotnet run --no-launch-profile
 DOCKER-COMPOSE ==> docker compose -f docker-compose.prod.yml up -d

 namespace Funkos.Infrastructure;
 
 public static class ApplicationInfraestructure
 {
     public static IServiceCollection AddApplicationServices(this IServiceCollection services, IConfiguration configuration)
     {
         // ... (Tus repositorios y servicios CRUD se quedan igual)
         #region Repositorios y Servicios CRUD
         services.AddScoped<IFunkoRepository, FunkoRepository>();
         services.AddScoped<ICategoriaRepository, CategoriaRepository>();
         services.AddScoped<IProveedorRepository, ProveedorRepository>();
         
         services.AddScoped<FunkoService>();
         services.AddScoped<ProveedorService>();
         services.AddScoped<CategoriaService>();
         #endregion
 
         #region VALIDACIONES
         services.AddValidatorsFromAssemblyContaining<ProveedorUpdateValidation>();
         #endregion
 
         #region Decorator
         services.AddScoped<IFunkoService>(sp => new CachedFunkoServiceDecorator(sp.GetRequiredService<FunkoService>(), sp.GetRequiredService<ICacheService>(), sp.GetRequiredService<ILogger<CachedFunkoServiceDecorator>>()));
         services.AddScoped<ICategoriaService>(sp => new CachedCategoriaServiceDecorator(sp.GetRequiredService<CategoriaService>(), sp.GetRequiredService<ICacheService>(), sp.GetRequiredService<ILogger<CachedCategoriaServiceDecorator>>()));
         services.AddScoped<IProveedorService>(sp => new CachedProveedorServiceDecorator(sp.GetRequiredService<ProveedorService>(), sp.GetRequiredService<ICacheService>(), sp.GetRequiredService<ILogger<CachedProveedorServiceDecorator>>()));
         #endregion
 
         #region POSTGRESQL
         var connectionString = configuration.GetConnectionString("DefaultConnection");
         services.AddDbContext<AppDbContext>(options => 
         {
             options.UseNpgsql(connectionString, npgsqlOptions =>
             {
                 // Si no estamos en desarrollo (es producción), activamos resiliencia ante cortes de red
                 // (Nota: necesitarás recibir 'IWebHostEnvironment env' o evaluarlo mediante el configuration o builder si se pasa, 
                 // pero como aquí solo tenemos IConfiguration, puedes leer el entorno así:)
             });
         });
         #endregion
 
         #region MONGO
         var settings = configuration.GetSection("MongoDbSettings").Get<MongoDbSettings>()!;
         services.AddSingleton(settings);
         services.AddSingleton<IMongoClient>(_ => new MongoClient(settings.ConnectionString));
         services.AddSingleton<MongoDbContext>();
         #endregion
 
         #region REDIS - CACHE
         // Leemos la connection string directamente de la configuración (que cambiará automáticamente según appsettings.Production.json o variables de entorno)
         var redisConnectionString = configuration["Redis:ConnectionString"]!;
         
         services.AddSingleton<IConnectionMultiplexer>(_ => ConnectionMultiplexer.Connect(redisConnectionString));
         services.AddSingleton<ICacheService, RedisCacheService>();
         services.Configure<CacheOptions>(configuration.GetSection(CacheOptions.SectionName));
         services.AddStackExchangeRedisCache(options =>
         {
             options.Configuration = redisConnectionString;
         });
         #endregion
 
         #region IMAGES
         services.AddScoped<IStorageService, FileSystemStorageService>();
         services.Configure<StorageSettings>(configuration.GetSection("Storage"));
 
         services.Configure<FormOptions>(options =>
         {
             options.MultipartBodyLengthLimit = 10 * 1024 * 1024; // 10 MB
         });
         #endregion
 
         #region IDENTITY Y JWT
         services.AddIdentityCore<User>(options =>
             {
                 options.Password.RequireDigit = true;           
                 options.Password.RequireLowercase = true;       
                 options.Password.RequireUppercase = true;       
                 options.Password.RequireNonAlphanumeric = true; 
                 options.Password.RequiredLength = 8;            
                 options.User.RequireUniqueEmail = true;         
                 options.Lockout.DefaultLockoutTimeSpan = TimeSpan.FromMinutes(15); 
                 options.Lockout.MaxFailedAccessAttempts = 5;    
                 options.ClaimsIdentity.UserIdClaimType = JwtRegisteredClaimNames.Sub;
             })
             .AddRoles<Role>()
             .AddSignInManager()
             .AddEntityFrameworkStores<AppDbContext>()
             .AddDefaultTokenProviders();
 
         services.AddAuthentication(options =>
             {
                 options.DefaultScheme = JwtBearerDefaults.AuthenticationScheme;                 
                 options.DefaultAuthenticateScheme = JwtBearerDefaults.AuthenticationScheme;     
                 options.DefaultChallengeScheme = JwtBearerDefaults.AuthenticationScheme;        
                 options.DefaultForbidScheme = JwtBearerDefaults.AuthenticationScheme;           
             })
             .AddJwtBearer(options =>
             {
                 options.MapInboundClaims = false;
                 
                 options.TokenValidationParameters = new TokenValidationParameters
                 {
                     ValidateIssuerSigningKey = true,
                     IssuerSigningKey = new SymmetricSecurityKey(
                         Encoding.UTF8.GetBytes(configuration["Jwt:Secret"] ?? throw new InvalidOperationException("JWT Secret no configurado"))),
                     ValidateIssuer = true,
                     ValidIssuer = configuration["Jwt:Issuer"],
                     ValidateAudience = true,
                     ValidAudience = configuration["Jwt:Audience"],
                     ValidateLifetime = true,
                     ClockSkew = TimeSpan.Zero,
                     RoleClaimType = "role"
                 };
             });
         #endregion
 
         #region AUTORIZACIONES
         services.AddAuthorizationBuilder()
             .SetDefaultPolicy(
                 new AuthorizationPolicyBuilder(JwtBearerDefaults.AuthenticationScheme)
                     .RequireAuthenticatedUser()
                     .Build())
             .AddPolicy("RequireAdmin", policy =>
                 policy.RequireAssertion(context =>
                         context.User.IsInRole("ADMIN") && context.User.HasClaim("department", "IT")))
             .AddPolicy("RequireUser", policy =>
                 policy.RequireRole("USER", "ADMIN"));
         #endregion
         
         #region SignalR y JWT
         services.AddSignalR().AddHubOptions<NotificacionesHub>(options =>
         {
             // Ocultamos errores detallados de SignalR en producción por seguridad
             options.EnableDetailedErrors = configuration.GetValue<bool>("SignalR:EnableDetailedErrors"); 
         })
         .AddStackExchangeRedis(redisConnectionString, options =>
         {
             options.Configuration.ChannelPrefix = "FunkosApp";
         });
         
         services.Configure<JwtBearerOptions>(JwtBearerDefaults.AuthenticationScheme, options =>
             {
                 options.Events = new JwtBearerEvents
                 {
                     OnMessageReceived = context =>
                     {
                         var accessToken = context.Request.Query["access_token"];
                         var path = context.HttpContext.Request.Path;
 
                         if (!string.IsNullOrEmpty(accessToken) && path.StartsWithSegments("/hubs"))
                         {
                             context.Token = accessToken;
                         }
                         return Task.CompletedTask;
                     }
                 };
             });
         #endregion
 
         #region CORS (Diferenciado por entorno)
         services.AddCors(options =>
         {
             // Política permisiva para desarrollo local
             options.AddPolicy("ClientesDev", policy =>
                 policy.WithOrigins("http://localhost:5175", "https://localhost:5175", "http://localhost:63342")
                     .AllowAnyHeader()
                     .AllowAnyMethod()
                     .AllowCredentials());
 
             // Política restrictiva para producción (Dominio real de la app)
             options.AddPolicy("ClientesProd", policy =>
                 policy.WithOrigins("https://miaplicacion.com")
                     .AllowAnyHeader()
                     .AllowAnyMethod()
                     .AllowCredentials());
         });
         #endregion
 
         #region RateLimit
         services.AddRateLimiter(options =>
         {
             options.AddPolicy("api", context =>
             {
                 var ip = context.Connection.RemoteIpAddress?.ToString() ?? "unknown";
                 return RateLimitPartition.GetFixedWindowLimiter(ip, _ => new FixedWindowRateLimiterOptions
                 {
                     PermitLimit = 100,
                     Window = TimeSpan.FromMinutes(1),
                     QueueProcessingOrder = QueueProcessingOrder.OldestFirst
                 });
             });
 
             options.AddPolicy("premium", context =>
             {
                 var userId = context.User.FindFirst("sub")?.Value ?? "anonymous";
                 return RateLimitPartition.GetFixedWindowLimiter(userId, _ => new FixedWindowRateLimiterOptions
                 {
                     PermitLimit = 1000,
                     Window = TimeSpan.FromMinutes(1)
                 });
             });
 
             options.OnRejected = async (context, cancellationToken) =>
             {
                 context.HttpContext.Response.StatusCode = 429;
                 await context.HttpContext.Response.WriteAsJsonAsync(new
                 {
                     error = "Demasiadas peticiones. Intenta de nuevo más tarde.",
                     retryAfter = context.Lease.TryGetMetadata(MetadataName.RetryAfter, out var retryAfter) ? retryAfter.TotalSeconds : (double?)null
                 }, cancellationToken);
             };
         });
         
         services.AddSingleton<RateLimitingState>();
         #endregion
 
         #region GRAPHQL con HotChocolate
         var graphqlServer = services
             .AddGraphQLServer()
             .AddQueryType<TiendaQuery>()
             .AddMutationType<TiendaMutation>()
             .AddType<FunkoType>()
             .AddType<CategoriaType>()
             .AddType<ProveedorType>()
             .AddSubscriptionType<Subscription>()
             .AddType<UploadType>()
             .AddInMemorySubscriptions()
             .AddAuthorization()
             .AddFiltering()
             .AddSorting()
             .ModifyOptions(options => options.RemoveUnreachableTypes = true)
             .AddProjections();
 
         // Control de errores e Introspección de GraphQL según configuración o entorno
         // (En producción deshabilitamos la introspección pública para mayor seguridad)
         var isDevelopment = configuration.GetValue<string>("ASPNETCORE_ENVIRONMENT") == "Development";
         
         graphqlServer.ModifyRequestOptions(opt =>
         {
             opt.IncludeExceptionDetails = isDevelopment;
         });
 
         if (!isDevelopment)
         {
             graphqlServer.DisableIntrospection();
         }
         #endregion
         
         return services;
     }
 }
